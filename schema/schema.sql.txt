-- Retail BI Engine Schema
-- Clean Analytics Table

CREATE TABLE superstore_clean (
    row_id INTEGER PRIMARY KEY,
    order_id TEXT,
    order_date TEXT,
    ship_date TEXT,
    ship_mode TEXT,
    customer_id TEXT,
    customer_name TEXT,
    segment TEXT,
    country TEXT,
    city TEXT,
    state TEXT,
    postal_code INTEGER,
    region TEXT,
    product_id TEXT,
    category TEXT,
    sub_category TEXT,
    product_name TEXT,
    sales REAL,
    quantity INTEGER,
    discount REAL,
    profit REAL
);




-- INSERTING THE DATA IN THE TABLE superstore_clean USING THE ORIGINAL TABLE superstore

INSERT INTO superstore_clean (
    row_id,
    order_id,
    order_date,
    ship_date,
    ship_mode,
    customer_id,
    customer_name,
    segment,

    region,

    product_name,

SELECT
    "Row ID",
