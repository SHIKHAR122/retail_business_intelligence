--Identify High-Sales but Low-Margin Categories
WITH category_metrics AS (
    SELECT
        category,
        SUM(sales) AS total_sales,
        SUM(profit) AS total_profit,
        SUM(profit) * 1.0 / SUM(sales) AS profit_margin
    FROM superstore_clean
    GROUP BY category
),

overall_avg AS (
    SELECT
        AVG(total_sales) AS avg_sales,
        AVG(profit_margin) AS avg_margin
    FROM category_metrics
)

SELECT
    c.category,
    c.total_sales,
    c.total_profit,
    ROUND(c.profit_margin, 4) AS profit_margin
FROM category_metrics c
CROSS JOIN overall_avg o
WHERE 
    c.total_sales > o.avg_sales
    AND c.profit_margin < o.avg_margin
ORDER BY c.total_sales DESC;






--Detect Loss-Making Subcategories


WITH subcategory_metrics AS (

        sub_category,
        SUM(sales) AS total_sales,
        SUM(profit) AS total_profit,
        SUM(profit) * 1.0 / SUM(sales) AS profit_margin
    FROM superstore_clean
    GROUP BY sub_category
)

SELECT
    sub_category,
    total_sales,
    total_profit,
    ROUND(profit_margin, 4) AS profit_margin,
    DENSE_RANK() OVER (ORDER BY total_profit ASC) AS loss_rank
FROM subcategory_metrics
WHERE total_profit < 0
ORDER BY total_profit ASC;

--Identify Highest & Lowest Performing Regions
  

WITH regional_sales AS (

        region,
        SUM(sales) AS total_sales
    FROM superstore_clean
    GROUP BY region
)

SELECT 
    region,
    total_sales,
    CASE 
        WHEN total_sales = (SELECT MAX(total_sales) FROM regional_sales)

    FROM superstore_clean

),
overall_avg AS (

        AVG(profit_margin) AS avg_margin
    FROM category_summary

SELECT 
    c.category,
    c.total_sales,
    c.total_profit,
    c.profit_margin,
    CASE 
        WHEN c.total_sales > o.avg_sales 
             AND c.profit_margin < o.avg_margin
            THEN 'High Sales - Low Margin (Risk)'
        WHEN c.total_sales < o.avg_sales 
             AND c.profit_margin > o.avg_margin
            THEN 'Low Sales - High Margin (Opportunity)'
        ELSE 'Normal'
    END AS performance_type
FROM category_summary c, overall_avg o
ORDER BY c.total_sales DESC;

--Comparison Of  Growth Across Segments


WITH annual_metrics AS (
    SELECT
        segment,
        strftime('%Y', order_date) AS order_year,
        SUM(sales)  AS total_sales,
        SUM(profit) AS total_profit
    FROM superstore_clean
    GROUP BY
        segment,
        strftime('%Y', order_date)
),

bounds AS (
    SELECT
        segment,
        order_year,
        total_sales,
        total_profit,

        FIRST_VALUE(total_sales) OVER (
            PARTITION BY segment
            ORDER BY order_year
        ) AS first_sales,

        LAST_VALUE(total_sales) OVER (
            PARTITION BY segment
            ORDER BY order_year
            ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
        ) AS last_sales,

        FIRST_VALUE(total_profit) OVER (
            PARTITION BY segment
            ORDER BY order_year
        ) AS first_profit,

        LAST_VALUE(total_profit) OVER (
            PARTITION BY segment
            ORDER BY order_year
            ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
        ) AS last_profit

    FROM annual_metrics
)

SELECT DISTINCT
    segment,

    first_sales,
    last_sales,

    ROUND((last_sales - first_sales) * 1.0 / first_sales * 100, 2)
        AS overall_sales_growth_pct,

    ROUND((POWER(last_sales * 1.0 / first_sales, 1.0/3) - 1) * 100, 2)
        AS sales_cagr_pct,

    first_profit,
    last_profit,

    ROUND((last_profit - first_profit) * 1.0 / first_profit * 100, 2)
        AS overall_profit_growth_pct,

    ROUND((POWER(last_profit * 1.0 / first_profit, 1.0/3) - 1) * 100, 2)
        AS profit_cagr_pct,

    ROUND(
        (POWER(last_sales * 1.0 / first_sales, 1.0/3) - 1) -
        (POWER(last_profit * 1.0 / first_profit, 1.0/3) - 1),
        4
    ) AS growth_gap_cagr

FROM bounds
ORDER BY segment;

--Month-over-Month (MoM) Growth Rate



WITH monthly_sales AS (
    SELECT
        strftime('%Y', order_date) AS order_year,
        strftime('%m', order_date) AS order_month,
        SUM(sales) AS monthly_sales
    FROM superstore_clean
    GROUP BY
        strftime('%Y', order_date),
        strftime('%m', order_date)
)

SELECT
    order_year,
    order_month,
    monthly_sales,
    
    LAG(monthly_sales) OVER (
        ORDER BY order_year, order_month
    ) AS previous_month_sales,

    (
        (monthly_sales - LAG(monthly_sales) OVER (
            ORDER BY order_year, order_month
        ))
        /
        LAG(monthly_sales) OVER (
            ORDER BY order_year, order_month
        )
    ) * 100 AS mom_growth_percent

FROM monthly_sales
ORDER BY order_year, order_month;






--Running Cumulative Sales


WITH monthly_sales AS (

--Seasonal pattern Detection 



WITH monthly_sales AS (
    SELECT
        strftime('%Y', order_date) AS order_year,
        strftime('%m', order_date) AS order_month,
        SUM(sales) AS monthly_sales
    FROM superstore_clean
    GROUP BY
        strftime('%Y', order_date),
        strftime('%m', order_date)
),

year_avg AS (
    SELECT
        order_year,
        AVG(monthly_sales) AS yearly_avg
    FROM monthly_sales
    GROUP BY order_year
)

SELECT
    m.order_year,
    m.order_month,
    ROUND(m.monthly_sales / y.yearly_avg, 3) AS normalized_sales

    ON m.order_year = y.order_year

--Monthly Profit Margin Trend



WITH monthly_metrics AS (
    SELECT
        strftime('%Y', order_date) AS order_year,
        strftime('%m', order_date) AS order_month,
        SUM(sales) AS monthly_sales,
        SUM(profit) AS monthly_profit
    FROM superstore_clean
    GROUP BY
        strftime('%Y', order_date),
        strftime('%m', order_date)
),

margin_calc AS (
    SELECT
        order_year,
        order_month,
        ROUND((monthly_profit * 100.0) / monthly_sales, 2) AS profit_margin_percent
    FROM monthly_metrics
)

SELECT
    order_year,
    order_month,
    profit_margin_percent,
    
    LAG(profit_margin_percent) OVER (
        ORDER BY order_year, order_month
    ) AS previous_month_margin,
    
    ROUND(
        profit_margin_percent -
        LAG(profit_margin_percent) OVER (
            ORDER BY order_year, order_month
        ),
    2) AS mom_margin_change

FROM margin_calc
ORDER BY order_year, order_month;







--Highest And Lowest Sales



WITH sales_metrices AS (
    SELECT 
        strftime("%Y", order_date) AS order_year,
        strftime("%m", order_date) AS order_month,
        SUM(sales) AS monthly_sales
    FROM superstore_clean
    GROUP BY 
        strftime("%Y", order_date),
        strftime("%m", order_date)
),

sales_rank AS (
    SELECT
        order_year,
        order_month,
        monthly_sales,
        ROW_NUMBER() OVER (
            PARTITION BY order_year 
            ORDER BY monthly_sales DESC
        ) AS highest_rank,
        ROW_NUMBER() OVER (
            PARTITION BY order_year 
            ORDER BY monthly_sales ASC
        ) AS lowest_rank
    FROM sales_metrices
)

SELECT 
    order_year,
    order_month,
    CASE 
        WHEN highest_rank = 1 THEN 'Highest Month'
        WHEN lowest_rank = 1 THEN 'Lowest Month'
    END AS month_type
FROM sales_rank
WHERE highest_rank = 1
   OR lowest_rank = 1
ORDER BY order_year, month_type;

--Region-Level Dependency Risk Detection

WITH region_sales AS (
    SELECT 
        strftime("%Y", order_date) AS order_year,
        region,
        SUM(sales) AS region_sales
    FROM superstore_clean
    GROUP BY strftime("%Y", order_date), region
),

overall_sales AS (
    SELECT
        strftime("%Y", order_date) AS order_year,
        SUM(sales) AS overall_sales
    FROM superstore_clean
    GROUP BY strftime("%Y", order_date)
),

region_metrics AS (
    SELECT
        r.order_year,
        r.region,
        r.region_sales,
        (r.region_sales * 100.0) / o.overall_sales AS region_percentage,
        ROW_NUMBER() OVER (
            PARTITION BY r.order_year
            ORDER BY r.region_sales DESC
        ) AS region_rank,
        SUM((r.region_sales * 100.0) / o.overall_sales)
            OVER (
                PARTITION BY r.order_year
                ORDER BY r.region_sales DESC
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
            ) AS cumulative_percentage
    FROM region_sales r
    JOIN overall_sales o
        ON r.order_year = o.order_year
),

risk_summary AS (
    SELECT
        order_year,

        -- Top 1 share
        MAX(CASE WHEN region_rank = 1 THEN region_percentage END) AS top1_share,

        -- Top 2 combined share
        SUM(CASE WHEN region_rank <= 2 THEN region_percentage END) AS top2_share,

        -- Number of regions needed to reach 80%
        COUNT(CASE WHEN cumulative_percentage <= 80 THEN 1 END) AS regions_to_80

    FROM region_metrics
    GROUP BY order_year
)

SELECT
    order_year,
    ROUND(top1_share, 2) AS top1_share,
    ROUND(top2_share, 2) AS top2_share,
    regions_to_80,

    CASE
        WHEN top1_share > 50 THEN 'High Dependency Risk'
        WHEN top2_share > 80 THEN 'Moderate Dependency Risk'
        ELSE 'Low Dependency Risk'
    END AS dependency_risk_level

FROM risk_summary
ORDER BY order_year;
